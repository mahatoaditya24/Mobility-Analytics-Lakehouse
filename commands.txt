# =============================================================================
# Mobility Analytics: CLI Command Reference Sheet
# =============================================================================

# 1. Start Infrastructure
docker compose up -d

# 2. Create Kafka Topic (if running manually)
docker exec -it kafka /opt/kafka/bin/kafka-topics.sh \
  --create \
  --if-not-exists \
  --topic traffic-topic \
  --bootstrap-server kafka:9092 \
  --partitions 3 \
  --replication-factor 1

# 3. Run Producer (from host)
pip install -r requirements.txt
python producer/traffic_producer.py --rate 2.5 --dirty-ratio 0.20

# 4. Submit Bronze Streaming Pipeline (Kafka -> Bronze Delta Table)
docker exec -it spark-worker /opt/spark/bin/spark-submit \
  --conf spark.jars.ivy=/tmp/.ivy \
  --packages io.delta:delta-spark_2.12:3.2.0,org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.1 \
  /opt/spark-apps/bronze_layer.py

# 5. Submit Silver Pipeline (Bronze -> DQ Validation -> Silver Clean + DLQ Quarantine)
docker exec -it spark-worker /opt/spark/bin/spark-submit \
  --conf spark.jars.ivy=/tmp/.ivy \
  --packages io.delta:delta-spark_2.12:3.2.0,org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.1 \
  /opt/spark-apps/silver_layer.py

# 6. Submit Gold Pipeline (Silver -> Star Schema: Fact & Dimensions)
docker exec -it spark-worker /opt/spark/bin/spark-submit \
  --conf spark.jars.ivy=/tmp/.ivy \
  --packages io.delta:delta-spark_2.12:3.2.0,org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.1 \
  /opt/spark-apps/gold_layer.py

# 7. Run Aggregations Job (Hourly Zone Congestion Rollups)
docker exec -it spark-worker /opt/spark/bin/spark-submit \
  --conf spark.jars.ivy=/tmp/.ivy \
  --packages io.delta:delta-spark_2.12:3.2.0 \
  /opt/spark-apps/aggregations.py

# 8. Interactive Spark-SQL Console (with Hive Metastore and Delta Catalog)
docker exec -it spark-worker /opt/spark/bin/spark-sql \
  --packages io.delta:delta-spark_2.12:3.2.0 \
  --conf spark.jars.ivy=/tmp/.ivy \
  --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension \
  --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog \
  --conf spark.sql.catalogImplementation=hive \
  --conf spark.hadoop.hive.metastore.uris=thrift://hive-metastore:9083 \
  --conf spark.sql.warehouse.dir=/opt/spark/warehouse

# 9. Start Spark Thrift Server (for external JDBC/ODBC BI tools)
docker exec -it spark-worker /opt/spark/sbin/start-thriftserver.sh \
  --master spark://spark-master:7077 \
  --conf spark.sql.extensions=io.delta.sql.DeltaSparkSessionExtension \
  --conf spark.sql.catalog.spark_catalog=org.apache.spark.sql.delta.catalog.DeltaCatalog \
  --conf spark.sql.catalogImplementation=hive \
  --conf spark.hadoop.hive.metastore.uris=thrift://hive-metastore:9083 \
  --conf spark.sql.warehouse.dir=/opt/spark/warehouse

# 10. Run Streamlit Dashboard (from host)
streamlit run dashboard/app.py

# 11. Run Unit Tests with Coverage
pytest tests/ -v --cov=producer --cov-report=term-missing
